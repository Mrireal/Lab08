-- BLOCKS WORLD para Manito --
Abstracción simple: sin coordenadas, sin cinemática, sin colisiones.

-- Predicados --

1. ontable(X)      # X está sobre la mesa
2. on(X, Y)        # X está sobre Y (X encima, Y debajo)
3. clear(X)        # X está libre (nada encima)
4. holding(X)      # Garra sostiene X
5. handempty       # Garra está vacía

-- Acciones (4) --

1. pickup(X)
   Precondiciones:
     - clear(X)
     - ontable(X)
     - handempty
   Efectos:
     - holding(X)
     - ¬handempty
     - ¬ontable(X)
     - ¬clear(X)

2. putdown(X)
   Precondiciones:
     - holding(X)
   Efectos:
     - ontable(X)
     - handempty
     - ¬holding(X)
     - clear(X)

3. stack(X, Y)
   Precondiciones:
     - holding(X)
     - clear(Y)
   Efectos:
     - on(X, Y)
     - clear(X)
     - ¬holding(X)
     - handempty
     - ¬clear(Y)

4. unstack(X, Y)
   Precondiciones:
     - on(X, Y)
     - clear(X)
     - handempty
   Efectos:
     - holding(X)
     - clear(Y)
     - ¬on(X, Y)
     - ¬clear(X)
     - ¬handempty

-- Escena inicial (siempre igual) --
- blue, red, green sobre la mesa y separados
- Todos libres (clear)
- Garra vacía (handempty)

-- Metas --
Goal A: green → red → blue (arriba → abajo)
Goal B: blue → red → green (orden invertido)
