-- MUNDO DE BLOQUES para Manito --
Abstracción simple: sin coordenadas, sin cinemática, sin colisiones.

-- Predicados --

1. sobre_mesa(X)    # X está sobre la mesa
2. sobre(X, Y)      # X está sobre Y (X encima, Y debajo)
3. libre(X)         # X está libre (nada encima)
4. sosteniendo(X)   # Garra sostiene X
5. mano_vacia       # Garra está vacía

-- Acciones (4) --

1. tomar(X)
   Precondiciones:
     - libre(X)
     - sobre_mesa(X)
     - mano_vacia
   Efectos:
     - sosteniendo(X)
     - ¬mano_vacia
     - ¬sobre_mesa(X)
     - ¬libre(X)

2. soltar(X)
   Precondiciones:
     - sosteniendo(X)
   Efectos:
     - sobre_mesa(X)
     - mano_vacia
     - ¬sosteniendo(X)
     - libre(X)

3. apilar(X, Y)
   Precondiciones:
     - sosteniendo(X)
     - libre(Y)
   Efectos:
     - sobre(X, Y)
     - libre(X)
     - mano_vacia
     - ¬sosteniendo(X)
     - ¬libre(Y)

4. desapilar(X, Y)
   Precondiciones:
     - sobre(X, Y)
     - libre(X)
     - mano_vacia
   Efectos:
     - sosteniendo(X)
     - libre(Y)
     - ¬sobre(X, Y)
     - ¬libre(X)
     - ¬mano_vacia

-- Escena inicial (siempre igual) --
- azul, rojo, verde sobre la mesa y separados
- Todos libres (sin nada encima)
- Garra vacía (mano_vacia)

-- Metas --
Meta A: verde → rojo → azul (arriba → abajo)
Meta B: azul → rojo → verde (orden invertido)
